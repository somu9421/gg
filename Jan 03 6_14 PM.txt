function calculateTip(){
    const billAmount = document.getElementById("billAmount").value;
    const tipPercentage = document.getElementById("tipPercentage").value;
    const tipAmount = (billAmount*tipPercentage)/100;
    const totalAmount = parseFloat(billAmount) + tipAmount;

    document.getElementById("tipAmount").innerText = tipAmount.toFixed(2);
    document.getElementById("totalAmount").innerText = totalAmount.toFixed(2);
}

function calculateBMI(){
    const gender=document.getElementById('gender').value;
    const weight=parseFloat(document.getElementById('weight').value);
    const height=parseFloat(document.getElementById('height').value);

    const resultDiv=document.getElementById('result');

    resultDiv.textContent='';

    if(!gender || !weight || !height){
        resultDiv.textContent='Please fill out all fields.';
        return; 
    }

    const bmi=(weight/((height/100)**2)).toFixed(2);

    let class1='' ;
    let col='';
    let dd='';

    if(gender=='male'){
           dd='men';
    }else{
        dd='women';
    }
      

    if(bmi<18.5){
        class1='Underweight';
        col='blue';
        
        // resultDiv.innerHTML=`<span class="${col}">Your BMI is ${bmi} ("${class1}") For ${dd},a BMI between 18.5 and 24.9 is considered healthy.</span>  `;

    }else if(bmi>=18.5 && bmi<24.9){
        class1='Normal weight';
        col='green'; 
        

        // resultDiv.innerHTML=`<span class="${col}">Your BMI is ${bmi} ("${class1}") For ${dd},a BMI between 18.5 and 24.9 is considered healthy.</span> `;

    }else if(bmi>=25 && bmi<29.9){
        class1='Overweight';
        col='red';
        // resultDiv.innerHTML=`<span class="${col}">Your BMI is ${bmi} ("${class1}") For ${dd},a BMI between 18.5 and 24.9 is considered healthy.</span> `;

    }else{
        class1='Obese';
        col='orange';
        // resultDiv.innerHTML=`<span class="${col}">Your BMI is ${bmi} ("${class1}") For ${dd},a BMI between 18.5 and 24.9 is considered healthy.</span>  `;

    }
     resultDiv.className=`${col}`;
    //  document.getElementById('result').className="${col}";
    resultDiv.innerHTML=`Your BMI is ${bmi} (${class1}) For ${dd},a BMI between 18.5 and 24.9 is considered healthy.`;


}
