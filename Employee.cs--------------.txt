                return BadRequest();

            _context.Employees.Add(employee);

        {
            if(employee == null || employee.DepartmentID <= 0 || int.Parse(employee.JoiningYear) <= 2024 || id < 0)

            objEmployee.JoiningYear = employee.JoiningYear;

            _context.SaveChanges();

            return Ok(objEmployee);
        }

        [HttpGet("filter")]

            {
                return BadRequest();

            var query = _context.Employees.AsQueryable();

            query = expression.Equals("desc",StringComparison.OrdinalIgnoreCase)?query.OrderByDescending(e => EF.Property<object>(e,name)):query.OrderBy(e => EF.Property<object>(e,name));

            if(list == null)
                return NoContent();

            return Ok(list);
        }

    {

        public DbSet<Employee> Employees {get; set;}
        public DbSet<Department> Departments {get;set;}
        public EmployeeApplicationDbContext(DbContextOptions<EmployeeApplicationDbContext> options):base(options){}

        protected override void OnModelCreating(ModelBuilder modelBuilder){
            base.OnModelCreating(modelBuilder);

            modelBuilder.Entity<Employee>().
            HasOne(e => e.Department).
            WithMany(d => d.Employees).
            HasForeignKey(e => e.DepartmentID).
            OnDelete(DeleteBehavior.Cascade);
        }

    }
}
